11 LOOP Example

  2      v_pokemon Pokemon%ROWTYPE;

  6      INTO v_pokemon

 11                     DBMS_OUTPUT.PUT_LINE(v_num||' Pokemon ID: ' || v_pokemon.pokemon_id);
 12     end if;
 13     if v_num=2 then
 14                     DBMS_OUTPUT.PUT_LINE(v_num||' Name: ' || v_pokemon.pokemon_name);
