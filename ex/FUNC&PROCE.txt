SQL> CREATE OR REPLACE FUNCTION reverse_string(p_str IN VARCHAR2)
  2      RETURN VARCHAR2
  3  IS
  4      v_rev VARCHAR2(4000);
  5  BEGIN
  6      FOR i IN REVERSE 1..LENGTH(p_str) LOOP
