SELF JOIN TO FIND THE MASTER OF TRAINERS IN THE TABLE TRAINER
=============================================================

SQL> SELECT C1.trainer_name AS TRAINER_NAME,
  2         C1.trainer_id AS TRAINER_ID,
  3         C2.master_name AS MASTER_NAME,
  4         C2.master_id AS MASTER_ID
  5  FROM Trainer C1
  6  JOIN Trainer C2
  7  ON C1.trainer_id = C2.master_id
  8  AND C1.trainer_name = C2.master_name;
