TO FIND TRAINER WHOSE AGE IS ABOVE THE AVERAGE AGE USING WITH CLAUSE
===================================================================

SQL> WITH AvgAge AS (
  2    SELECT AVG(age) AS avg_age FROM Trainer
  3  )
  4  SELECT trainer_name, age
  5  FROM Trainer
  6  WHERE age > (SELECT avg_age FROM AvgAge);

------------------------------ ----------

TO DISPLAY TRAINERS AGE IS ABOVE THE AVERAGE AND WHO HAS THE SAME NO OF BADGE AS MORE THAN 1 TRAINERS
=====================================================================================================

SQL> WITH AvgAge AS (
  2    SELECT AVG(age) AS avg_age FROM Trainer
  3  ),
  4  BadgeCount AS (
  5    SELECT badge, COUNT(*) AS badge_count
  6    FROM Trainer
  7    GROUP BY badge
  8  )
  9  SELECT C1.trainer_name, C1.age, C2.badge, C2.badge_count
 10  FROM Trainer C1
 11  JOIN BadgeCount C2 ON C1.badge = C2.badge
 12  WHERE C1.age > (SELECT avg_age FROM AvgAge)
 13  AND C2.badge_count > 1;

TRAINER_NAME                          AGE      BADGE BADGE_COUNT
------------------------------ ---------- ---------- -----------
Ash                                    25          2           2
Misty                                  25          2           2
